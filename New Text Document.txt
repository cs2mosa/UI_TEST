from huggingface_hub import snapshot_download
import urllib.request, os

# 1. Download/link the dataset directly to /workspace/isafety_bench
print('[*] Resolving dataset into /workspace/isafety_bench ...')
dest = '/workspace/isafety_bench'
snapshot_download(
    repo_id='raiyaanabdullah/isafety-bench',
    repo_type='dataset',
    local_dir=dest
)

# 2. Download annotations directly into that same folder
base_url = 'https://raw.githubusercontent.com/iSafetyBench/data/main'
for fname in ['annotations_hazard.json', 'annotations_normal.json', 'actions.txt']:
    print(f'[*] Downloading {fname} ...')
    urllib.request.urlretrieve(f'{base_url}/{fname}', os.path.join(dest, fname))

print('[+] Ready! Dataset and annotations are in: /workspace/isafety_bench')
"
